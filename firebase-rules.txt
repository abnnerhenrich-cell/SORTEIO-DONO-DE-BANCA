rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // Sorteios de dezenas
    match /sorteios_dezenas/{docId} {
      allow read, write: if true;
    }
    match /participantes_dezenas/{docId} {
      allow read, write: if true;
    }
    match /ganhadores_dezenas/{docId} {
      allow read, write: if true;
    }

    // Sorteios de centenas
    match /sorteios_centenas/{docId} {
      allow read, write: if true;
    }
    match /participantes_centenas/{docId} {
      allow read, write: if true;
    }
    match /ganhadores_centenas/{docId} {
      allow read, write: if true;
    }

    // Aparência / URL de cadastro
    match /config/{docId} {
      allow read, write: if true;
    }
  }
}
